% PDF-only preamble (pandoc -H). Reader feedback: at 11pt, mono vs serif
% alone is hard to tell apart for older eyes. Set inline code (and plain
% verbatim blocks) in a dark blue -- the PicoMite-manual convention -- so
% commands stand out from body text by colour, not just letterform.
% Syntax-highlighted (```python) blocks keep their tango colours.
%
% The colour is given numerically, not as a named colour: inline code in
% section titles reaches the running page headers, which the book class
% passes through \MakeUppercase -- a colour NAME gets uppercased into an
% undefined one, but digits survive.
\usepackage{xcolor}
\let\pcTexttt\texttt
\renewcommand{\texttt}[1]{\textcolor[RGB]{0,70,160}{\pcTexttt{#1}}}
\makeatletter
\def\verbatim@font{\ttfamily\color[RGB]{0,70,160}}
\makeatother

% Reader feedback: code listings pasted from the PDF arrived flush-left --
% their indentation was gone. Cause: xelatex stores verbatim spaces as
% horizontal positioning, not as characters, so copy/paste (and the REPL)
% never see them. Fix: make fancyvrb emit a real space glyph (\char"20) for
% every code-block space. It has the correct monospace width, so the page
% looks identical, but the spaces are now extractable and listings paste with
% their indentation intact. (\XeTeXgenerateactualtext alone does NOT fix this
% -- the spaces are glue between runs, which ActualText spans don't capture.)
\makeatletter
\def\FV@Space{\char"20\relax}
\makeatother

% Full-bleed cover page, WITHOUT changing the page geometry.
%
% This used to drop the margins with \newgeometry for the cover and hand
% them back with \restoregeometry. The image bled correctly, but the
% restore did not reach the page that followed: the first contents page
% came out with the cover's zero-margin footskip, its folio printed
% 0.7pt from the paper edge (841.2 of A4's 841.9) where every other page
% ends at 762, and a reader noticed it looked wrong beside the rest.
%
% eso-pic hangs the image on the shipout of THIS page only, behind the
% text, in paper coordinates. No dimension changes, so nothing can leak:
% the contents page after it is laid out exactly like every other page.
% \null gives the page a body to ship; the image is placed from the
% paper's lower left at \paperwidth x \paperheight, which is the cover's
% own aspect, so it fills the sheet with no letterbox slivers.
%
% NOT the titlepage ENVIRONMENT, and that is the whole point. book.cls
% ends `titlepage' with \setcounter{page}\@ne -- it puts the page
% counter back to 1 -- so with pandoc's \frontmatter (roman) the cover
% was page i and the first contents page was page i AGAIN. hyperref
% writes those numbers into the PDF as page labels, so a print dialog
% offered two pages called i and printing "page i" fetched the cover as
% well. Reported by a reader trying to print the contents.
%
% Numbering the cover i and running on -- contents ii, iii, ... -- gives
% every sheet a label of its own, which is what a printer needs. The
% cover shows no number itself: it is a full-bleed image with an empty
% page style.
\usepackage{graphicx}
\usepackage{eso-pic}
\renewcommand{\maketitle}{%
  \thispagestyle{empty}%
  \AddToShipoutPictureBG*{%
    \AtPageLowerLeft{%
      \includegraphics[width=\paperwidth,height=\paperheight]{cover.png}}}%
  \null
  \clearpage
}
